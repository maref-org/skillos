```python
# 新建: tools/web_fetch.py
import httpx
from bs4 import BeautifulSoup
from readability import Document

class WebFetchTool:
    """网页内容获取与智能提取工具"""
    
    def __init__(self, timeout: int = 30):
        self.client = httpx.AsyncClient(
            timeout=timeout,
            follow_redirects=True,
            headers={"User-Agent": "SkillOS-WebFetch/1.0"}
        )
    
    async def fetch(
        self,
        url: str,
        extract_mode: str = "full",
        max_length: int = 10000
    ) -> dict:
        """获取并解析网页内容"""
        response = await self.client.get(url)
        response.raise_for_status()
        
        html = response.text
        soup = BeautifulSoup(html, "html.parser")
        
        result = {
            "title": soup.title.string if soup.title else "",
            "metadata": self._extract_metadata(soup),
            "content": "",
            "links": []
        }
        
        if extract_mode == "article":
            doc = Document(html)
            result["content"] = doc.summary()[:max_length]
        elif extract_mode == "links":
            result["links"] = [a["href"] for a in soup.find_all("a", href=True)]
        elif extract_mode == "metadata":
            result["content"] = self._extract_text(soup, max_length)
        else:  # full
            result["content"] = self._extract_text(soup, max_length)
            result["links"] = [a["href"] for a in soup.find_all("a", href=True)]
        
        return result
    
    def _extract_metadata(self, soup: BeautifulSoup) -> dict:
        """提取页面元数据"""
        meta = {}
        for tag in soup.find_all("meta"):
            name = tag.get("name") or tag.get("property")
            if name:
                meta[name] = tag.get("content", "")
        return meta
    
    def _extract_text(self, soup: BeautifulSoup, max_length: int) -> str:
        """智能文本提取 - 去除脚本和样式"""
        for script in soup(["script", "style"]):
            script.decompose()
        text = soup.get_text(separator="\n", strip=True)
        return text[:max_length]
```

```python
# 新建: kernel/session_isolator.py
import asyncio
import os
import shutil
import tempfile
from dataclasses import dataclass
from pathlib import Path

@dataclass
class IsolationContext:
    worktree_path: str
    sandbox_id: str
    resource_limits: dict
    session_id: str

class SessionIsolator:
    """Session 隔离执行管理器"""
    
    def __init__(self, base_dir: str = "/tmp/skillos-isolation"):
        self.base_dir = Path(base_dir)
        self.base_dir.mkdir(parents=True, exist_ok=True)
        self.active_sessions: dict[str, IsolationContext] = {}
    
    async def create_isolated_session(
        self,
        session_id: str,
        repo_path: str | None = None,
        resource_limits: dict | None = None
    ) -> IsolationContext:
        """创建隔离执行会话"""
        sandbox_id = f"sandbox_{session_id}"
        worktree_path = self.base_dir / sandbox_id
        
        # 1. 创建隔离工作目录
        worktree_path.mkdir(parents=True, exist_ok=True)
        
        # 2. 如果有源码，创建 git worktree 隔离副本
        if repo_path:
            await self._create_git_worktree(repo_path, worktree_path)
        
        # 3. 设置资源限制
        limits = resource_limits or {
            "max_memory_mb": 2048,
            "max_cpu_percent": 50,
            "max_disk_mb": 1024,
            "timeout_seconds": 300
        }
        
        ctx = IsolationContext(
            worktree_path=str(worktree_path),
            sandbox_id=sandbox_id,
            resource_limits=limits,
            session_id=session_id
        )
        self.active_sessions[session_id] = ctx
        return ctx
    
    async def execute_in_isolation(
        self,
        session_id: str,
        command: str,
        timeout: int | None = None
    ) -> dict:
        """在隔离环境中执行命令"""
        ctx = self.active_sessions.get(session_id)
        if not ctx:
            raise ValueError(f"Session {session_id} not found")
        
        timeout = timeout or ctx.resource_limits["timeout_seconds"]
        
        proc = await asyncio.create_subprocess_exec(
            *command.split(),
            cwd=ctx.worktree_path,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        
        try:
            stdout, stderr = await asyncio.wait_for(
                proc.communicate(),
                timeout=timeout
            )
            return {
                "stdout": stdout.decode(),
                "stderr": stderr.decode(),
                "exit_code": proc.returncode
            }
        except asyncio.TimeoutError:
            proc.kill()
            return {"error": "Execution timeout", "exit_code": -1}
    
    async def cleanup_session(self, session_id: str) -> bool:
        """清理隔离会话"""
        ctx = self.active_sessions.pop(session_id, None)
        if ctx:
            shutil.rmtree(ctx.worktree_path, ignore_errors=True)
            return True
        return False
    
    async def _create_git_worktree(self, repo_path: str, worktree_path: Path):
        """创建 git worktree 隔离副本"""
        branch_name = f"skillos-isolation-{worktree_path.name}"
        proc = await asyncio.create_subprocess_exec(
            "git", "worktree", "add", str(worktree_path), "-b", branch_name,
            cwd=repo_path
        )
        await proc.wait()
```

#### 3.3.2 跨会话状态持久化

**目标**: 实现跨会话的 memory/tasks 持久化

```python
# 新建: kernel/state_persistor.py
import json
import sqlite3
from dataclasses import asdict, dataclass
from datetime import datetime
from pathlib import Path
from typing import Any

@dataclass
class PersistentState:
    session_id: str
    state_key: str
    state_value: Any
    created_at: str
    updated_at: str
    ttl_seconds: int = 0  # 0 = 永久

class StatePersistor:
    """跨会话状态持久化存储"""
    
    def __init__(self, db_path: str = "data/skillos_state.db"):
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self._init_db()
    
    def _init_db(self):
        """初始化状态存储表"""
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS session_states (
                    session_id TEXT NOT NULL,
                    state_key TEXT NOT NULL,
                    state_value TEXT NOT NULL,
                    created_at TEXT NOT NULL,
                    updated_at TEXT NOT NULL,
                    ttl_seconds INTEGER DEFAULT 0,
                    PRIMARY KEY (session_id, state_key)
                )
            """)
            conn.execute("""
                CREATE TABLE IF NOT EXISTS task_memory (
                    task_id TEXT PRIMARY KEY,
                    session_id TEXT NOT NULL,
                    context TEXT NOT NULL,
                    status TEXT NOT NULL DEFAULT 'pending',
                    result TEXT,
                    created_at TEXT NOT NULL,
                    completed_at TEXT
                )
            """)
    
    def save_state(self, state: PersistentState) -> bool:
        """保存会话状态"""
        now = datetime.now().isoformat()
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                INSERT OR REPLACE INTO session_states 
                (session_id, state_key, state_value, created_at, updated_at, ttl_seconds)
                VALUES (?, ?, ?, ?, ?, ?)
            """, (
                state.session_id,
                state.state_key,
                json.dumps(state.state_value),
                now if state.created_at == "" else state.created_at,
                now,
                state.ttl_seconds
            ))
        return True
    
    def load_state(self, session_id: str, state_key: str) -> Any | None:
        """加载会话状态"""
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.execute(
                "SELECT state_value, ttl_seconds FROM session_states WHERE session_id=? AND state_key=?",
                (session_id, state_key)
            )
            row = cursor.fetchone()
            if not row:
                return None
            
            state_value = json.loads(row[0])
            ttl = row[1]
            
            # 检查 TTL
            if ttl > 0:
                cursor2 = conn.execute(
                    "SELECT updated_at FROM session_states WHERE session_id=? AND state_key=?",
                    (session_id, state_key)
                )
                updated_at = datetime.fromisoformat(cursor2.fetchone()[0])
                if (datetime.now() - updated_at).total_seconds() > ttl:
                    return None  # 过期
            
            return state_value
    
    def save_task_result(self, task_id: str, session_id: str, result: dict, context: dict):
        """保存任务执行结果到 memory"""
        now = datetime.now().isoformat()
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                INSERT OR REPLACE INTO task_memory 
                (task_id, session_id, context, status, result, created_at, completed_at)
                VALUES (?, ?, ?, 'completed', ?, ?, ?)
            """, (
                task_id,
                session_id,
                json.dumps(context),
                json.dumps(result),
                now,
                now
            ))
    
    def get_task_history(self, session_id: str, limit: int = 10) -> list[dict]:
        """获取会话任务历史"""
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.execute(
                "SELECT task_id, context, status, result, created_at FROM task_memory WHERE session_id=? ORDER BY created_at DESC LIMIT ?",
                (session_id, limit)
            )
            return [
                {
                    "task_id": row[0],
                    "context": json.loads(row[1]),
                    "status": row[2],
                    "result": json.loads(row[3]) if row[3] else None,
                    "created_at": row[4]
                }
                for row in cursor.fetchall()
            ]
```

#### 3.3.3 定时任务调度

**目标**: 实现 cron 风格的定时任务调度

```python
# 新建: kernel/task_scheduler.py
import asyncio
import json
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path
from typing import Any, Callable

import croniter

@dataclass
class ScheduledTask:
    task_id: str
    name: str
    cron_expression: str
    handler: str  # skill_id 或函数名
    params: dict[str, Any]
    enabled: bool = True
    last_run: str | None = None
    next_run: str | None = None
    success_count: int = 0
    failure_count: int = 0

class TaskScheduler:
    """定时任务调度器"""
    
    def __init__(self, config_path: str = "data/scheduled_tasks.json"):
        self.config_path = Path(config_path)
        self.tasks: dict[str, ScheduledTask] = {}
        self._running = False
        self._load_tasks()
    
    def _load_tasks(self):
        """加载已配置的定时任务"""
        if self.config_path.exists():
            with open(self.config_path) as f:
                data = json.load(f)
                for task_data in data:
                    self.tasks[task_data["task_id"]] = ScheduledTask(**task_data)
    
    def _save_tasks(self):
        """保存任务配置"""
        with open(self.config_path, "w") as f:
            json.dump([
                {
                    "task_id": t.task_id,
                    "name": t.name,
                    "cron_expression": t.cron_expression,
                    "handler": t.handler,
                    "params": t.params,
                    "enabled": t.enabled,
                    "last_run": t.last_run,
                    "next_run": t.next_run,
                    "success_count": t.success_count,
                    "failure_count": t.failure_count
                }
                for t in self.tasks.values()
            ], f, indent=2)
    
    def add_task(
        self,
        task_id: str,
        name: str,
        cron_expression: str,
        handler: str,
        params: dict[str, Any] | None = None
    ) -> ScheduledTask:
        """添加定时任务"""
        # 验证 cron 表达式
        try:
            cron = croniter.croniter(cron_expression, datetime.now())
            next_run = cron.get_next(datetime)
        except (ValueError, TypeError) as e:
            raise ValueError(f"Invalid cron expression: {e}")
        
        task = ScheduledTask(
            task_id=task_id,
            name=name,
            cron_expression=cron_expression,
            handler=handler,
            params=params or {},
            next_run=next_run.isoformat()
        )
        self.tasks[task_id] = task
        self._save_tasks()
        return task
    
    def remove_task(self, task_id: str) -> bool:
        """移除定时任务"""
        if task_id in self.tasks:
            del self.tasks[task_id]
            self._save_tasks()
            return True
        return False
    
    async def start(self):
        """启动调度器"""
        self._running = True
        asyncio.create_task(self._scheduler_loop())
    
    async def stop(self):
        """停止调度器"""
        self._running = False
    
    async def _scheduler_loop(self):
        """调度循环"""
        while self._running:
            now = datetime.now()
            for task in self.tasks.values():
                if not task.enabled:
                    continue
                
                # 检查是否需要执行
                if task.next_run and datetime.fromisoformat(task.next_run) <= now:
                    await self._execute_task(task)
                
                # 更新下次执行时间
                cron = croniter.croniter(task.cron_expression, datetime.now())
                task.next_run = cron.get_next(datetime).isoformat()
            
            self._save_tasks()
            await asyncio.sleep(60)  # 每分钟检查一次
    
    async def _execute_task(self, task: ScheduledTask):
        """执行单个任务"""
        try:
            # 根据 handler 调用对应技能或函数
            result = await self._invoke_handler(task.handler, task.params)
            task.success_count += 1
            task.last_run = datetime.now().isoformat()
        except Exception as e:
            task.failure_count += 1
            print(f"Task {task.task_id} failed: {e}")
```

```python
# 新建: protocol/remote_control.py
import asyncio
import json
from dataclasses import dataclass, field
from typing import Any
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel

class RemoteCommand(BaseModel):
    command: str
    session_id: str | None = None
    params: dict[str, Any] = {}
    timeout: int = 300

class RemoteControlResponse(BaseModel):
    status: str
    result: Any | None = None
    error: str | None = None

@dataclass
class BridgeSession:
    session_id: str
    remote_url: str
    api_key: str
    status: str = "disconnected"
    last_heartbeat: float = 0.0

class RemoteBridge:
    """远程桥接控制器"""
    
    def __init__(self):
        self.active_bridges: dict[str, BridgeSession] = {}
        self.router = APIRouter(prefix="/remote", tags=["remote-control"])
        self._setup_routes()
    
    def _setup_routes(self):
        """设置远程控制 API 路由"""
        self.router.post("/command")(self.execute_remote_command)
        self.router.post("/bridge/connect")(self.connect_bridge)
        self.router.post("/bridge/disconnect")(self.disconnect_bridge)
        self.router.get("/bridge/status")(self.get_bridge_status)
        self.router.get("/sessions")(self.list_sessions)
    
    async def execute_remote_command(self, cmd: RemoteCommand) -> RemoteControlResponse:
        """执行远程命令"""
        try:
            # 路由到对应会话或创建新会话
            session_id = cmd.session_id or await self._create_session()
            
            # 执行命令
            result = await self._dispatch_command(session_id, cmd.command, cmd.params, cmd.timeout)
            
            return RemoteControlResponse(status="success", result=result)
        except Exception as e:
            return RemoteControlResponse(status="error", error=str(e))
    
    async def connect_bridge(self, session_id: str, remote_url: str, api_key: str) -> dict:
        """连接远程桥接"""
        bridge = BridgeSession(
            session_id=session_id,
            remote_url=remote_url,
            api_key=api_key,
            status="connected"
        )
        self.active_bridges[session_id] = bridge
        return {"status": "connected", "session_id": session_id}
    
    async def disconnect_bridge(self, session_id: str) -> dict:
        """断开远程桥接"""
        bridge = self.active_bridges.pop(session_id, None)
        if bridge:
            bridge.status = "disconnected"
            return {"status": "disconnected"}
        raise HTTPException(status_code=404, detail="Bridge not found")
    
    async def get_bridge_status(self, session_id: str) -> dict:
        """获取桥接状态"""
        bridge = self.active_bridges.get(session_id)
        if not bridge:
            raise HTTPException(status_code=404, detail="Bridge not found")
        return {
            "session_id": bridge.session_id,
            "status": bridge.status,
            "remote_url": bridge.remote_url,
            "last_heartbeat": bridge.last_heartbeat
        }
    
    async def list_sessions(self) -> list[dict]:
        """列出所有活跃会话"""
        return [
            {
                "session_id": b.session_id,
                "status": b.status,
                "remote_url": b.remote_url
            }
            for b in self.active_bridges.values()
        ]
    
    async def _create_session(self) -> str:
        """创建新会话"""
        import uuid
        session_id = str(uuid.uuid4())
        return session_id
    
    async def _dispatch_command(
        self,
        session_id: str,
        command: str,
        params: dict,
        timeout: int
    ) -> Any:
        """分发命令到对应处理器"""
        # 根据命令类型路由到不同处理器
        command_handlers = {
            "skill_execute": self._handle_skill_execute,
            "session_create": self._handle_session_create,
            "session_close": self._handle_session_close,
            "state_query": self._handle_state_query,
        }
        handler = command_handlers.get(command)
        if not handler:
            raise ValueError(f"Unknown command: {command}")
        return await handler(session_id, params, timeout)
    
    async def _handle_skill_execute(self, session_id: str, params: dict, timeout: int) -> dict:
        """处理技能执行命令"""
        # 调用 skill_kernel 执行技能
        from kernel.skill_kernel import SkillKernel
        kernel = SkillKernel()
        result = await kernel.execute(
            skill_name=params.get("skill_name"),
            input_data=params.get("input_data", {}),
            session_id=session_id,
            timeout=timeout
        )
        return result
    
    async def _handle_session_create(self, session_id: str, params: dict, timeout: int) -> dict:
        """处理会话创建命令"""
        from kernel.session_manager import SessionManager
        manager = SessionManager()
        session = await manager.create_session(user_id=params.get("user_id", "remote"))
        return {"session_id": session.session_id}
    
    async def _handle_session_close(self, session_id: str, params: dict, timeout: int) -> dict:
        """处理会话关闭命令"""
        from kernel.session_manager import SessionManager
        manager = SessionManager()
        await manager.close_session(session_id)
        return {"status": "closed"}
    
    async def _handle_state_query(self, session_id: str, params: dict, timeout: int) -> dict:
        """处理状态查询命令"""
        from kernel.state_persistor import StatePersistor
        store = StatePersistor()
        state = store.load_state(session_id, params.get("key", ""))
        return {"state": state}
```
